(entries) => {
  if (entries[0].isIntersecting) {
      // DOM 렌더링 없이 브라우저 캐시에만 적재
      const img = new Image();
      img.decoding = 'async'; // 메인 스레드 블로킹 방지

      observer.disconnect();
    }
  },
